# The +cpu local version is not decoration: it exists ONLY on the PyTorch CPU
# index, so pinning it stops pip resolving `torch==2.8.0` from PyPI instead,
# which on Linux bundles the CUDA runtime — roughly 900 MB of wheels this
# service cannot use. The encoder never calls .to(DEVICE); _load_net maps to
# cpu and both synth passes are plain tensor math.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.8.0+cpu

# The versions the checkpoints were rendered with (requirements.txt at the repo
# root). The audio must not change because the host did.
numpy==2.0.2
scipy==1.13.1
soundfile==0.13.1

# gradio brings its own fastapi/uvicorn/pydantic, which is why they are not
# pinned here: pinning them against gradio's range is how you get a resolver
# conflict on a host you cannot debug interactively.
gradio==6.26.0
