# The pin is PLAIN 2.8.0, but the wheel that lands is the CPU one.
#
# Hugging Face validates this file as text: with `torch==2.8.0+cpu` a ZeroGPU
# Space refuses to start with "torch version in requirements.txt is not
# compatible with ZeroGPU. Supported versions: ... 2.8.0" — it does not
# recognise the local version. Dropping `+cpu` satisfies that check, and pip
# still installs the CPU build, because PEP 440 sorts `2.8.0+cpu` ABOVE plain
# `2.8.0` and both satisfy `==2.8.0`, so the extra index wins on version order.
#
# Verified, not reasoned about — `pip download` for cp312/manylinux_2_28_x86_64
# with these two lines fetches torch-2.8.0+cpu-...manylinux_2_28_x86_64.whl
# (183.9 MB). If that ever stops being true the symptom is a ~900 MB CUDA wheel
# in the build log, which this service cannot use: the encoder never calls
# .to(DEVICE), _load_net maps to cpu and both synth passes are plain tensor math.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.8.0

# The versions the checkpoints were rendered with (requirements.txt at the repo
# root). The audio must not change because the host did.
numpy==2.0.2
scipy==1.13.1
soundfile==0.13.1

# gradio brings its own fastapi/uvicorn/pydantic, which is why they are not
# pinned here: pinning them against gradio's range is how you get a resolver
# conflict on a host you cannot debug interactively.
gradio==6.26.0

# is the platform's own shim and the platform is what it has to match.
spaces
